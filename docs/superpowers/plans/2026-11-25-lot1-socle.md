# Lot 1 — Socle : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** poser le socle serveur de l'outil d'interview utilisateurs : base D1, session admin par mot de passe, proxy Anthropic verrouillé, catalogue de modèles à jour, routage par URL.

**Architecture :** le code serveur partagé vit dans `server/` (hors de `functions/`, pour ne pas créer de routes parasites) et les types communs au navigateur et au serveur dans `shared/`. Les Pages Functions de `functions/api/` ne font qu'appeler `server/`. Le front passe à React Router et appelle l'API en chemins relatifs, via un proxy Vite en développement, ce qui rend le cookie de session « same-origin ».

**Tech stack :** Cloudflare Pages Functions, D1, `@anthropic-ai/sdk`, React 18, React Router 6, Zustand, Vitest + Miniflare.

**Spec :** `docs/superpowers/specs/2026-11-25-interview-utilisateurs-design.md` (sections 3, 7, 8, 9 et lot 1 de la section 10).

**Hors de ce lot :** écrans de projets, logiciels, questionnaires, parcours répondant, MCP, analyses. Le schéma D1 complet est créé ici mais seules les tables sont posées, sans service.

```ts
import { readFileSync, readdirSync } from 'node:fs'
import { join } from 'node:path'
import { Miniflare } from 'miniflare'

const DOSSIER = join(process.cwd(), 'migrations')

/** Base D1 en mémoire avec toutes les migrations appliquées. Appeler `fermer()` en fin de test. */
export async function creerBaseDeTest() {
  const mf = new Miniflare({
    modules: true,
    script: 'export default { fetch() { return new Response(null) } }',
    d1Databases: ['DB'],
  })
  const db = await mf.getD1Database('DB')
  for (const fichier of readdirSync(DOSSIER).filter(f => f.endsWith('.sql')).sort()) {
    const sql = readFileSync(join(DOSSIER, fichier), 'utf8')
      .split('\n')
      .filter(ligne => !ligne.trim().startsWith('--'))
      .join('\n')
    for (const instruction of sql.split(';').map(s => s.trim()).filter(Boolean)) {
      await db.prepare(instruction).run()
    }
  }
  return { db, fermer: () => mf.dispose() }
}
```

```sql
  maj_le TEXT NOT NULL

  id TEXT PRIMARY KEY,

  maj_le TEXT NOT NULL

CREATE TABLE projet_logiciels (
  projet_id TEXT NOT NULL REFERENCES projets(id) ON DELETE CASCADE,
  logiciel_id TEXT NOT NULL REFERENCES logiciels(id) ON DELETE CASCADE,
  PRIMARY KEY (projet_id, logiciel_id)

CREATE TABLE documents (
  id TEXT PRIMARY KEY,

CREATE TABLE questionnaires (
  id TEXT PRIMARY KEY,

CREATE TABLE analyses (
  id TEXT PRIMARY KEY,
```

```ts
import { erreur } from './http'

export const NOM_COOKIE = 'vs_admin'
export const DUREE_SESSION_S = 60 * 60 * 24 * 30

const encodeur = new TextEncoder()

async function hmacHex(secret: string, message: string): Promise<string> {
  const cle = await crypto.subtle.importKey(
    'raw', encodeur.encode(secret), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign'],
  )
  const signature = await crypto.subtle.sign('HMAC', cle, encodeur.encode(message))
  return [...new Uint8Array(signature)].map(o => o.toString(16).padStart(2, '0')).join('')
}

function egalEnTempsConstant(a: string, b: string): boolean {
  if (a.length !== b.length) return false
  let difference = 0
  for (let i = 0; i < a.length; i++) difference |= a.charCodeAt(i) ^ b.charCodeAt(i)
  return difference === 0
}

/** Valeur du cookie : `<échéance en ms>.<signature>`. */
export async function creerSession(secret: string, maintenantMs: number): Promise<string> {
  const echeance = String(maintenantMs + DUREE_SESSION_S * 1000)
  return `${echeance}.${await hmacHex(secret, `session:${echeance}`)}`
}

export async function sessionValide(
  secret: string, valeur: string | null, maintenantMs: number,
): Promise<boolean> {
  if (!valeur) return false
  const morceaux = valeur.split('.')
  if (morceaux.length !== 2) return false
  const [echeance, signature] = morceaux
  if (!/^\d+$/.test(echeance) || Number(echeance) <= maintenantMs) return false
  return egalEnTempsConstant(signature, await hmacHex(secret, `session:${echeance}`))
}

/** Compare deux mots de passe sans fuite de durée : on compare leurs empreintes. */
export async function motDePasseValide(secret: string, saisi: string, attendu: string): Promise<boolean> {
  const [a, b] = await Promise.all([hmacHex(secret, `mdp:${saisi}`), hmacHex(secret, `mdp:${attendu}`)])
  return egalEnTempsConstant(a, b)
}

export function lireCookie(requete: Request, nom: string): string | null {
  const enTete = requete.headers.get('Cookie')
  if (!enTete) return null
  for (const morceau of enTete.split(';')) {
    const [cle, ...reste] = morceau.trim().split('=')
    if (cle === nom) return reste.join('=')
  }
  return null
}

export function enTeteCookie(valeur: string, dureeS: number): string {
  return `${NOM_COOKIE}=${valeur}; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=${dureeS}`
}

/** Renvoie `null` si la requête porte une session admin valide, sinon la réponse de refus. */
export async function exigerAdmin(
  requete: Request, env: { SESSION_SECRET: string }, maintenantMs: number = Date.now(),
): Promise<Response | null> {
  if (!env.SESSION_SECRET) return erreur('SESSION_SECRET non configuré.', 500)
  const valide = await sessionValide(env.SESSION_SECRET, lireCookie(requete, NOM_COOKIE), maintenantMs)
  return valide ? null : erreur('Authentification requise.', 401)
}
```

```ts
import Anthropic from '@anthropic-ai/sdk'
import { CATALOGUE_SECOURS, DEFAUTS, type Catalogue, type Etape } from '../shared/modeles'

export interface ModeleApi {
  id: string
  display_name: string
  created_at: string
}

const DUREE_CACHE_MS = 24 * 60 * 60 * 1000

// Cache en mémoire de l'isolat : au mieux 24 h, perdu au redémarrage, ce qui suffit ici.
let cache: { expire: number; catalogue: Catalogue } | null = null

export function viderCache(): void {
  cache = null
}

export function construireCatalogue(liste: ModeleApi[]): Catalogue {
  const modeles = liste
    .filter(m => m.id.startsWith('claude-'))
    .sort((a, b) => b.created_at.localeCompare(a.created_at))
    .map(m => ({ id: m.id, nom: m.display_name }))
  if (modeles.length === 0) return CATALOGUE_SECOURS

  const connus = new Set(modeles.map(m => m.id))
  const choisir = (etape: Etape) => (connus.has(DEFAUTS[etape]) ? DEFAUTS[etape] : modeles[0].id)
  return {
    modeles,
    defauts: { cadrage: choisir('cadrage'), interview: choisir('interview'), analyse: choisir('analyse') },
  }
}

export async function obtenirCatalogue(
  lister: () => Promise<ModeleApi[]>, maintenantMs: number = Date.now(),
): Promise<Catalogue> {
  if (cache && cache.expire > maintenantMs) return cache.catalogue
  try {
    const catalogue = construireCatalogue(await lister())
    cache = { expire: maintenantMs + DUREE_CACHE_MS, catalogue }
    return catalogue
  } catch {
    return CATALOGUE_SECOURS
  }
}

export async function listerDepuisAnthropic(cleApi: string): Promise<ModeleApi[]> {
  const client = new Anthropic({ apiKey: cleApi })
  const modeles: ModeleApi[] = []
  for await (const modele of client.models.list()) {
    modeles.push({ id: modele.id, display_name: modele.display_name, created_at: modele.created_at })
  }
  return modeles
}
```

```ts
export async function sessionActive(): Promise<boolean> {
  try {
    return (await fetch('/api/admin/session')).ok
  } catch {
    return false
  }
}

export async function connexion(motDePasse: string): Promise<void> {
  const reponse = await fetch('/api/admin/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ motDePasse }),
  })
  if (!reponse.ok) {
    const corps = await reponse.json().catch(() => ({})) as { error?: { message?: string } }
    throw new Error(corps.error?.message ?? `Erreur ${reponse.status}`)
  }
}

export async function deconnexion(): Promise<void> {
  await fetch('/api/admin/logout', { method: 'POST' })
}
```

```tsx
import { useEffect, useState, type ReactNode } from 'react'
import { LoginScreen } from '../admin/LoginScreen'
import { Spinner } from '../components/ui/Spinner'
import { sessionActive } from '../lib/adminApi'

type Etat = 'verification' | 'connecte' | 'anonyme'

export function RequireAdmin({ children }: { children: ReactNode }) {
  const [etat, setEtat] = useState<Etat>('verification')

  useEffect(() => {
    let actif = true
    sessionActive().then(ok => { if (actif) setEtat(ok ? 'connecte' : 'anonyme') })
    return () => { actif = false }
  }, [])

  if (etat === 'verification') {
    return (
      <div className="min-h-screen bg-paper flex items-center justify-center">
        <Spinner size="lg" />
      </div>
    )
  }
  if (etat === 'anonyme') return <LoginScreen onSuccess={() => setEtat('connecte')} />
  return <>{children}</>
}
```
